function selectManual(key,label,group,s){
  const all=byGroup(group), val=s[key]||'';
  return `<div class="field"><label>${esc(label)}</label><select data-field="${esc(key)}">${option('','— Select from 2015 manual —',val)}${all.map(x=>option(x.code,`${x.code} — ${x.description} (unit ${x.unit}, M-${x.pdfPage-422})`,val)).join('')}${val&&!all.find(x=>x.code===val)?option(val,'Legacy value: '+val,val):''}</select></div>`;
}

 const title=type==='miscRows'?'Miscellaneous improvements':'Special features and adjustments';
 let out=`<h3>${title}</h3><p class="hint">Add one line per feature. Manual schedule references are shown; verify the applicable rate, units and calculation.</p><div class="manual-lines" data-linegroup="${type}">`;
 rows.forEach((r,i)=>{let val=r.code||'';out+=`<div class="line-row"><select data-line="${type}" data-index="${i}" data-key="code">${option('','Select improvement or feature',val)}${options.map(x=>option(x.value,x.label,val)).join('')}${val&&!options.some(x=>x.value===val)?option(val,val,val):''}</select><input type="number" step="any" min="0" placeholder="Quantity" data-line="${type}" data-index="${i}" data-key="quantity" value="${esc(r.quantity??'')}"><input type="text" placeholder="Units / size" data-line="${type}" data-index="${i}" data-key="units" value="${esc(r.units||'')}"><input type="text" placeholder="Notes / verified rate" data-line="${type}" data-index="${i}" data-key="notes" value="${esc(r.notes||'')}"><button type="button" class="small danger" data-remove="${type}" data-index="${i}">Remove</button></div>`});
 out+=`</div><button type="button" class="small secondary" data-add="${type}">+ Add ${type==='miscRows'?'improvement':'feature'}</button>`;return out;
}
function upgradedEditor(s){
 const base=groups.filter(([name])=>!['Building construction','Adjustments / other features'].includes(name));
 const old=editorHTML(base,s); // original function copied below before override
 return old+`<h3>Building construction · manual codes</h3>`+
 ['improvementType','occupancy','buildingClass','yearBuilt','length','width','stories','wallHeight'].map(k=>{const labels={improvementType:'Improvement type code',occupancy:'Occupancy code',buildingClass:'Construction class',yearBuilt:'Year built',length:'Length (ft)',width:'Width (ft)',stories:'Stories',wallHeight:'Wall height (ft)'};return k==='buildingClass'?field(labels[k],k,s[k],{list:['AA','A','B','C','D','E','F','D+','D-','C+','C-']}):simple(labels[k],k,s)}).join('')+
 selectManual('foundation','Foundation','Foundation',s)+selectManual('exterior','Exterior wall','Exterior wall',s)+selectManual('roof','Roof construction','Roof type',s)+selectManual('roofMaterial','Roof covering','Roof covering',s)+selectManual('floor','Floor','Floor',s)+selectManual('interior','Interior finish','Interior finish',s)+selectManual('plumbing','Plumbing category','Plumbing category',s)+
 `<h3>Additional construction adjustments</h3>`+field('Structural framing','framing',s.framing||'')+field('Party wall dimensions / adjustment','partyWall',s.partyWall||'')+field('Stall width','stallWidth',s.stallWidth||'')+field('Heating / air conditioning','heating',s.heating||'')+field('Appendages / dimension notes','appendages',s.appendages||'',{textarea:true})+
 addRowForm('miscRows',s)+addRowForm('featureRows',s)+
 `<h3>Existing freeform notes (retained)</h3>`+field('Previously entered miscellaneous notes','misc',s.misc||'',{textarea:true})+field('Previously entered feature notes','otherFeatures',s.otherFeatures||'',{textarea:true});
}

 const banner=document.createElement('div');banner.className='upgrade-actions';banner.innerHTML=`<p><b>Review-first PRC:</b> Enter values above, then select Generate Review. The structured output does not update on every keystroke. Existing records remain autosaved in this browser.</p><button type="button" class="primary" id="generateReview">Generate PRC review</button> <button type="button" class="secondary" id="clearReview">Return to entry</button>`;root.prepend(banner);
 root.querySelector('#generateReview').onclick=()=>{approvedSampleId=current().id;renderCanvas();toast('Review generated from saved entry data')};root.querySelector('#clearReview').onclick=()=>{approvedSampleId=null;renderCanvas()};
};
const originalSetSampleField=setSampleField;

function descriptor(code){const item=D.codes.find(x=>x.code===code);return item?`${item.code} — ${item.description} (unit ${item.unit})`:code||'—'}
function lineLabel(r){if(!r.code)return '(unselected)';let [type,i]=r.code.split(':');let x=type==='rate'?D.miscItems[+i]:type==='schedule'?D.miscCategories[+i]:D.specialFeatures[+i];return x?.name||r.code}
function lineRows(s,key){return(s[key]||[]).map(r=>`<tr><td>${esc(lineLabel(r))}</td><td>${esc(r.quantity)}</td><td>${esc(r.units)}</td><td>${esc(r.notes)}</td></tr>`).join('')||'<tr><td colspan="4">None entered</td></tr>'}
function cell(label,value){return `<div class="transcript-cell"><small>${esc(label)}</small><b>${esc(value??'—')}</b></div>`}
function cleanPRC(s,which,print){
 let rows=which==='front'?[['County',study().county],['Sample',s.num],['Parcel',s.parcel],['Owner',s.owner],['Grantee',s.grantee],['Address',s.address],['Account',s.account],['Acreage',s.acreage],['Assessment class',s.class],['Appraiser',s.appraiser],['Date',s.sampleDate],['Sale price',s.salePrice],['Land value',s.landValue],['Current improvement cost',money(currentCost(s))],['2015 base-year cost',s.manualBaseCost],['Sample index',num(costIndex(s))]]:[['Sample',s.num],['Improvement type',s.improvementType],['Occupancy',s.occupancy],['Construction class',s.buildingClass],['Built',s.yearBuilt],['Length (ft)',s.length],['Width (ft)',s.width],['Base area (SF)',n(s.length)*n(s.width)],['Stories',s.stories],['Wall height',s.wallHeight],['Foundation',descriptor(s.foundation)],['Exterior',descriptor(s.exterior)],['Roof',descriptor(s.roof)],['Roof material',descriptor(s.roofMaterial)],['Floor',descriptor(s.floor)],['Interior',descriptor(s.interior)],['Plumbing',descriptor(s.plumbing)],['Framing',s.framing],['Party walls',s.partyWall],['Stall width',s.stallWidth],['HVAC',s.heating],['Appendages',s.appendages],['Depreciation (%)',s.depreciation],['Manual base cost',s.manualBaseCost]];
 let extra=which==='back'?`<h3>Miscellaneous improvements</h3><table class="transcript-table"><thead><tr><th>Improvement</th><th>Qty</th><th>Units</th><th>Notes</th></tr></thead><tbody>${lineRows(s,'miscRows')}</tbody></table><h3>Special features and adjustments</h3><table class="transcript-table"><thead><tr><th>Feature</th><th>Qty</th><th>Units</th><th>Notes</th></tr></thead><tbody>${lineRows(s,'featureRows')}</tbody></table><h3>Legacy narrative</h3><p>${esc(s.misc||'—')} ${esc(s.otherFeatures||'')}</p>${s.media?.sketch?`<h3>Sketch</h3><img class="transcript-sketch" src="${s.media.sketch}" alt="Uploaded sketch">`:''}`:'';
 return `<div class="transcript ${print?'transcript-print':''}"><div class="transcript-head"><strong>ALABAMA PROPERTY RECORD — ${which.toUpperCase()} TRANSCRIPTION</strong><small>Review worksheet • Manual 2015 • Not an official preprinted PRC</small></div><div class="transcript-grid">${rows.map(([a,b])=>cell(a,b)).join('')}</div>${extra}<p class="transcript-note">Verify this transcription, detailed manual calculations, and source-document alignment before use. Values shown are not an independently validated appraisal.</p></div>`;
}
